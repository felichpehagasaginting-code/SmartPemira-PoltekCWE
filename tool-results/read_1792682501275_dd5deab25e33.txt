   128→export function PemilihPortal() {
   129→  const searchParams = useSearchParams();
   130→  const [phase, setPhase] = useState<Phase>("gate");
   131→  const [tokenChars, setTokenChars] = useState<string[]>(Array(6).fill(""));
   132→  const [gateError, setGateError] = useState<string | null>(null);
   133→  const [validating, setValidating] = useState(false);
   134→  const [session, setSession] = useState<KioskSession | null>(null);
   135→  const [isRevote, setIsRevote] = useState(false);
   136→  const [ballots, setBallots] = useState<BallotPayload[]>([]);
   137→  const [step, setStep] = useState(0);
   138→  const [selections, setSelections] = useState<Selections>({});
   139→  const [submitError, setSubmitError] = useState<string | null>(null);
   140→  const [remainingMs, setRemainingMs] = useState(0);
   141→  const [submittingMsg, setSubmittingMsg] = useState("Mengenkripsi & Mengacak Suara...");
   142→  const [receiptCode, setReceiptCode] = useState<string | null>(null);
   143→  const [voteVersion, setVoteVersion] = useState(1);
   144→  const [lockBusy, setLockBusy] = useState(false);
   145→  const activeTokenRef = useRef<string>("");
   146→  const autoTriedRef = useRef<string>("");
   147→
   148→  const current = ballots[step];
   149→
   150→  const fillToken = useCallback((token: string) => {
   151→    setTokenChars(token.slice(0, 6).padEnd(6, " ").split("").map((c) => (c === " " ? "" : c)));
   152→  }, []);
   153→
   154→  const validateToken = useCallback(async (token: string) => {
   155→    setValidating(true);
   156→    setGateError(null);
   157→    try {
   158→      const res = await fetch("/api/pemilih/validate", {
   159→        method: "POST",
   160→        headers: { "Content-Type": "application/json" },
   161→        body: JSON.stringify({ token }),
   162→      });
   163→      const json = await res.json();
   164→      if (!json.ok) {
   165→        setGateError(json.message ?? "Token tidak sah.");
   166→        return false;
   167→      }
   168→      setSession(json.session);
   169→      setBallots(json.ballots);
   170→      setIsRevote(Boolean(json.isRevote));
   171→      setVoteVersion(json.voteVersion ?? 1);
   172→      setSelections({});
   173→      setStep(0);
   174→      setPhase("ballot");
   175→      return true;
   176→    } catch {
   177→      setGateError("Koneksi ke server bilik suara gagal.");
   178→      return false;
   179→    } finally {
   180→      setValidating(false);
   181→    }
   182→  }, []);
   183→
   184→  // Auto-isi & auto-validasi token dari QR (?token=XXXXXX pada slip PPUR)
   185→  useEffect(() => {
   186→    const q = (searchParams.get("token") ?? "").trim().toUpperCase();
   187→    if (q.length === 6 && /^[A-Z0-9]{6}$/.test(q) && autoTriedRef.current !== q) {
   188→      autoTriedRef.current = q;
   189→      fillToken(q);
   190→      activeTokenRef.current = q;
   191→      void validateToken(q);
   192→    }
   193→  }, [searchParams, fillToken, validateToken]);
   194→
   195→  /** Hitung mundur TTL token — hanya vote PERTAMA (re-vote tanpa TTL) */
   196→  useEffect(() => {
   197→    if (!session || !session.expiresAt || (phase !== "ballot" && phase !== "review")) return;
   198→    const tick = () => {
   199→      const left = new Date(session.expiresAt!).getTime() - Date.now();
   200→      setRemainingMs(Math.max(0, left));
   201→      if (left <= 0) {
   202→        setPhase("gate");
   203→        setSession(null);
   204→        setBallots([]);
   205→        setSelections({});
   206→        setTokenChars(Array(6).fill(""));
   207→        setGateError("Token kedaluwarsa saat berada di bilik. Minta token baru di pos bantuan PPUR.");
   208→      }
   209→    };
   210→    tick();
   211→    const t = setInterval(tick, 1000);
   212→    return () => clearInterval(t);
   213→  }, [session, phase]);
   214→
   215→  const resetAll = useCallback(() => {
   216→    setPhase("gate");
   217→    setTokenChars(Array(6).fill(""));
   218→    setGateError(null);
   219→    setSession(null);
   220→    setBallots([]);
   221→    setStep(0);
   222→    setSelections({});
   223→    setSubmitError(null);
   224→    setReceiptCode(null);
   225→    setIsRevote(false);
   226→    activeTokenRef.current = "";
   227→  }, []);
   228→
   229→  const submitVote = useCallback(async () => {
   230→    if (!session) return;
   231→    setPhase("submitting");
   232→    setSubmittingMsg(isRevote ? "Mengganti Suara Anda..." : "Mengenkripsi & Mengacak Suara...");
   233→    try {
   234→      const res = await fetch("/api/pemilih/submit", {
   235→        method: "POST",
   236→        headers: { "Content-Type": "application/json" },
   237→        body: JSON.stringify({
   238→          token: activeTokenRef.current,
   239→          selections: {
   240→            presmaCandidateId: selections["PRESMA"],
   241→            dpmCandidateId: selections["DPM"],
   242→            himaCandidateId: selections["HIMA"] === "EMPTY" ? null : selections["HIMA"],
   243→          },
   244→        }),
   245→      });
   246→      const json = await res.json();
   247→      if (!json.ok) {
   248→        setSubmitError(json.message ?? "Gagal mengirim suara.");
   249→        setPhase("review");
   250→        return;
   251→      }
   252→      setReceiptCode(json.receiptCode ?? null);
   253→      setVoteVersion(json.voteVersion ?? 1);
   254→      setIsRevote(Boolean(json.isRevote));
   255→      setPhase("success");
   256→    } catch {
   257→      setSubmitError("Koneksi gagal saat mengirim suara. Suara tidak terkirim — coba lagi atau hubungi pos bantuan.");
   258→      setPhase("review");
   259→    }
   260→  }, [session, selections, isRevote]);
   261→
   262→  const lockVote = useCallback(async () => {
   263→    setLockBusy(true);
   264→    try {
   265→      const res = await fetch("/api/pemilih/lock", {
   266→        method: "POST",
   267→        headers: { "Content-Type": "application/json" },
   268→        body: JSON.stringify({ token: activeTokenRef.current }),
   269→      });
   270→      const json = await res.json();
   271→      if (json.ok) {
   272→        setPhase("locked");
   273→      } else {
   274→        setGateError(json.message ?? "Gagal mengunci suara.");
   275→      }
   276→    } catch {
   277→      setGateError("Koneksi gagal saat mengunci suara.");
   278→    } finally {
   279→      setLockBusy(false);
   280→    }
   281→  }, []);
   282→
   283→  const revoteAgain = useCallback(async () => {
   284→    // Buka bilik ulang dengan token yang sama — suara baru akan MENGGANTI yang lama
   285→    await validateToken(activeTokenRef.current);
   286→  }, [validateToken]);
   287→
   288→  /** Keyboard fisik juga didukung (desktop) */
   289→  const pressKey = useCallback((key: string) => {
   290→    setTokenChars((prev) => {
   291→      const idx = prev.findIndex((c) => c === "");
   292→      if (idx === -1) return prev;
   293→      const next = [...prev];
   294→      next[idx] = key;
   295→      return next;
   296→    });
   297→    setGateError(null);
   298→  }, []);
   299→
   300→  const backspace = useCallback(() => {
   301→    setTokenChars((prev) => {
   302→      const next = [...prev];
   303→      const idx = next.findIndex((c) => c === "");
   304→      const target = idx === -1 ? 5 : idx - 1;
   305→      if (target >= 0) next[target] = "";
   306→      return next;
   307→    });
   308→  }, []);
   309→
   310→  useEffect(() => {
   311→    if (phase !== "gate") return;
   312→    const handler = (e: KeyboardEvent) => {
   313→      if (/^[a-zA-Z0-9]$/.test(e.key)) pressKey(e.key.toUpperCase());
   314→      else if (e.key === "Backspace") backspace();
   315→    };
   316→    window.addEventListener("keydown", handler);
   317→    return () => window.removeEventListener("keydown", handler);
   318→  }, [phase, pressKey, backspace]);
   319→
   320→  const fullToken = tokenChars.join("");
   321→  const minutes = Math.floor(remainingMs / 60000);
   322→  const seconds = Math.floor((remainingMs % 60000) / 1000);
   323→  const timeCritical = remainingMs > 0 && remainingMs < 120_000;
   324→  const hasTtl = Boolean(session?.expiresAt);
   325→
   326→  const keypadKeys = useMemo(() => {
   327→    const digits = ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"];
   328→    const letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
   329→    return [...digits, ...letters];
   330→  }, []);
   331→
   332→  // ============================ GATE ============================
   333→  if (phase === "gate") {
   334→    return (
   335→      <div className="flex min-h-screen flex-col items-center justify-center px-4 py-10">
   336→        <div className="w-full max-w-md">
   337→          <div className="mb-7 text-center">
   338→            <div className="mx-auto mb-5 flex h-16 w-16 items-center justify-center rounded-[14px] border-2 border-[#141E19] bg-[#1B7F4E] shadow-[4px_4px_0px_#141E19]">
   339→              <Lock className="h-7 w-7 text-white" />
   340→            </div>
   341→            <h1 className="font-display text-3xl font-extrabold tracking-[-0.03em] md:text-4xl">
   342→              PORTAL PEMILIH <span className="text-[#1B7F4E]">PEMIRA ONLINE</span>
   343→            </h1>
   344→            <p className="mt-2 text-sm font-medium text-[#4A5851]">
   345→              Pemira Poltek CWE 100% Online — pilih dari perangkatmu sendiri
   346→            </p>
   347→          </div>
   348→
   349→          <div className="mb-4 flex items-center gap-2 rounded-[10px] border border-[#D8D3C8] bg-white px-4 py-3">
   350→            <ScanLine className="h-5 w-5 shrink-0 text-[#1B7F4E]" />
   351→            <p className="text-[12.5px] leading-relaxed text-[#4A5851]">
   352→              <strong className="text-[#141E19]">Cara masuk:</strong> scan <strong>QR code</strong> pada slip token dari pos
   353→              bantuan PPUR, atau ketik 6 karakter token di bawah.
   354→            </p>
   355→          </div>
   356→
   357→          {/* 6 Kotak Token */}
   358→          <div className="mb-5 flex justify-center gap-2" role="group" aria-label="Kotak input token">
   359→            {tokenChars.map((c, i) => (
   360→              <div
   361→                key={i}
   362→                className={`flex h-[68px] w-[52px] items-center justify-center rounded-[10px] border-2 bg-white font-mono text-2xl font-bold transition-shadow sm:h-[76px] sm:w-[58px] sm:text-3xl ${
   363→                  c ? "border-[#141E19] shadow-[3px_3px_0px_#141E19]" : "border-[#141E19]/40"
   364→                }`}
   365→                aria-label={`Karakter ${i + 1}: ${c || "kosong"}`}
   366→              >
   367→                {c}
   368→              </div>
   369→            ))}
   370→          </div>
   371→
   372→          {gateError && (
   373→            <div className="mx-auto mb-4 flex items-start gap-2 rounded-[10px] border-2 border-[#E05A47] bg-[#FDE8E5] px-4 py-3" role="alert">
   374→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-[#842029]" />
   375→              <p className="text-sm font-semibold text-[#842029]">{gateError}</p>
   376→            </div>
   377→          )}
   378→
   379→          {/* Keypad virtual (mobile-first) */}
   380→          <div className="tactile-card mx-auto p-4">
   381→            <div className="grid grid-cols-6 gap-1.5">
   382→              {keypadKeys.map((k) => (
   383→                <button
   384→                  key={k}
   385→                  type="button"
   386→                  onClick={() => pressKey(k)}
   387→                  className="h-10 rounded-lg border-2 border-[#141E19] bg-white font-mono text-sm font-bold shadow-[2px_2px_0px_#141E19] transition-all hover:-translate-y-0.5 hover:shadow-[3px_3px_0px_#141E19] active:translate-y-0.5 active:shadow-none sm:h-11"
   388→                >
   389→                  {k}
   390→                </button>
   391→              ))}
   392→              <button
   393→                type="button"
   394→                onClick={backspace}
   395→                className="col-span-2 flex h-10 items-center justify-center gap-1.5 rounded-lg border-2 border-[#141E19] bg-[#F2B705] text-sm font-bold shadow-[2px_2px_0px_#141E19] transition-all hover:-translate-y-0.5 active:translate-y-0.5 active:shadow-none sm:h-11"
   396→                aria-label="Hapus karakter terakhir"
   397→              >
   398→                <Delete className="h-4 w-4" /> HAPUS
   399→              </button>
   400→            </div>
   401→
   402→            <button
   403→              type="button"
   404→              disabled={fullToken.length !== 6 || validating}
   405→              onClick={() => {
   406→                activeTokenRef.current = fullToken;
   407→                validateToken(fullToken);
   408→              }}
   409→              className="tactile-btn mt-4 flex h-14 w-full items-center justify-center gap-2 text-base font-extrabold tracking-wide"
   410→            >
   411→              {validating ? "MEMVERIFIKASI TOKEN..." : "MASUK BILIK SUARA"}
   412→              {!validating && <ArrowRight className="h-5 w-5" />}
   413→            </button>
   414→          </div>
   415→
   416→          <p className="mt-5 text-center text-xs leading-relaxed text-[#7A8881]">
   417→            🔒 Zero-Knowledge Ballot — token anonim tanpa identitas. Pilihan Anda dipisahkan mutlak dari NIM.
   418→            Bisa ganti suara sampai pemungutan ditutup — suara terakhir yang dihitung.
   419→          </p>
   420→        </div>
   421→      </div>
   422→    );
   423→  }
   424→
   425→  // ============================ LOCKED (FINAL) ============================
   426→  if (phase === "locked") {
   427→    return (
   428→      <div className="flex min-h-screen flex-col items-center justify-center px-4 py-10 text-center">
   429→        <div className="anim-check mx-auto mb-6 flex h-24 w-24 items-center justify-center rounded-full border-[3px] border-[#141E19] bg-[#141E19] shadow-[6px_6px_0px_#1B7F4E]">
   430→          <Lock className="h-11 w-11 text-[#FEF5D6]" strokeWidth={3} />
   431→        </div>
   432→        <h1 className="font-display text-3xl font-extrabold tracking-[-0.03em] md:text-4xl">Suara Anda Terkunci Final</h1>
   433→        <p className="mt-3 max-w-md text-base leading-relaxed text-[#4A5851]">
   434→          Terima kasih! Pilihan Anda sudah <strong className="text-[#141E19]">DIKUNCI</strong> dan tidak dapat diganti lagi
   435→          oleh siapa pun. Ini melindungi suara Anda dari tekanan pihak lain.
   436→        </p>
   437→
   438→        {receiptCode && (
   439→          <div className="mt-7 w-full max-w-md rounded-[14px] border-2 border-[#141E19] bg-white p-5 text-left shadow-[6px_6px_0px_#141E19]">
   440→            <div className="flex items-center justify-between gap-2">
   441→              <p className="flex items-center gap-2 font-display text-sm font-extrabold uppercase tracking-wide">
   442→                <ReceiptText className="h-4 w-4 text-[#1B7F4E]" /> Struk Suara Digital — Versi {voteVersion}
   443→              </p>
   444→              <span className="rounded-full border border-[#141E19] bg-[#FEF5D6] px-2.5 py-0.5 text-[9.5px] font-extrabold text-[#8A5200]">
   445→                TERKUNCI FINAL
   446→              </span>
   447→            </div>
   448→            <p className="mt-3 font-mono text-[26px] font-extrabold tracking-[0.12em] text-[#141E19]">{receiptCode}</p>
   449→            <p className="mt-2 text-[12px] leading-relaxed text-[#4A5851]">
   450→              Simpan kode ini sebagai bukti suara tercatat di rantai hash. Cek kapan pun di Portal Publik → “Cek Struk Suara”.
   451→            </p>
   452→          </div>
   453→        )}
   454→
   455→        <button type="button" onClick={resetAll} className="mt-8 rounded-full border-2 border-[#141E19] bg-white px-6 py-2.5 text-sm font-bold shadow-[3px_3px_0px_#141E19] transition-all hover:-translate-y-0.5 active:translate-y-0.5 active:shadow-none">
   456→          Selesai
   457→        </button>
   458→      </div>
   459→    );
   460→  }
   461→
   462→  // ============================ SUCCESS ============================
   463→  if (phase === "success") {
   464→    return (
   465→      <div className="flex min-h-screen flex-col items-center justify-center px-4 py-10 text-center">
   466→        <div className="anim-check mx-auto mb-6 flex h-24 w-24 items-center justify-center rounded-full border-[3px] border-[#141E19] bg-[#1B7F4E] shadow-[6px_6px_0px_#141E19]">
   467→          <Check className="h-12 w-12 text-white" strokeWidth={3.5} />
   468→        </div>
   469→        <h1 className="font-display text-3xl font-extrabold tracking-[-0.03em] md:text-4xl">
