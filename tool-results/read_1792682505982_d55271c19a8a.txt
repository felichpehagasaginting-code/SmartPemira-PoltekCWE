   128→   128→export function PemilihPortal() {
   129→   129→  const searchParams = useSearchParams();
   130→   130→  const [phase, setPhase] = useState<Phase>("gate");
   131→   131→  const [tokenChars, setTokenChars] = useState<string[]>(Array(6).fill(""));
   132→   132→  const [gateError, setGateError] = useState<string | null>(null);
   133→   133→  const [validating, setValidating] = useState(false);
   134→   134→  const [session, setSession] = useState<KioskSession | null>(null);
   135→   135→  const [isRevote, setIsRevote] = useState(false);
   136→   136→  const [ballots, setBallots] = useState<BallotPayload[]>([]);
   137→   137→  const [step, setStep] = useState(0);
   138→   138→  const [selections, setSelections] = useState<Selections>({});
   139→   139→  const [submitError, setSubmitError] = useState<string | null>(null);
   140→   140→  const [remainingMs, setRemainingMs] = useState(0);
   141→   141→  const [submittingMsg, setSubmittingMsg] = useState("Mengenkripsi & Mengacak Suara...");
   142→   142→  const [receiptCode, setReceiptCode] = useState<string | null>(null);
   143→   143→  const [voteVersion, setVoteVersion] = useState(1);
   144→   144→  const [lockBusy, setLockBusy] = useState(false);
   145→   145→  const activeTokenRef = useRef<string>("");
   146→   146→  const autoTriedRef = useRef<string>("");
   147→   147→
   148→   148→  const current = ballots[step];
   149→   149→
   150→   150→  const fillToken = useCallback((token: string) => {
   151→   151→    setTokenChars(token.slice(0, 6).padEnd(6, " ").split("").map((c) => (c === " " ? "" : c)));
   152→   152→  }, []);
   153→   153→
   154→   154→  const validateToken = useCallback(async (token: string) => {
   155→   155→    setValidating(true);
   156→   156→    setGateError(null);
   157→   157→    try {
   158→   158→      const res = await fetch("/api/pemilih/validate", {
   159→   159→        method: "POST",
   160→   160→        headers: { "Content-Type": "application/json" },
   161→   161→        body: JSON.stringify({ token }),
   162→   162→      });
   163→   163→      const json = await res.json();
   164→   164→      if (!json.ok) {
   165→   165→        setGateError(json.message ?? "Token tidak sah.");
   166→   166→        return false;
   167→   167→      }
   168→   168→      setSession(json.session);
   169→   169→      setBallots(json.ballots);
   170→   170→      setIsRevote(Boolean(json.isRevote));
   171→   171→      setVoteVersion(json.voteVersion ?? 1);
   172→   172→      setSelections({});
   173→   173→      setStep(0);
   174→   174→      setPhase("ballot");
   175→   175→      return true;
   176→   176→    } catch {
   177→   177→      setGateError("Koneksi ke server bilik suara gagal.");
   178→   178→      return false;
   179→   179→    } finally {
   180→   180→      setValidating(false);
   181→   181→    }
   182→   182→  }, []);
   183→   183→
   184→   184→  // Auto-isi & auto-validasi token dari QR (?token=XXXXXX pada slip PPUR)
   185→   185→  useEffect(() => {
   186→   186→    const q = (searchParams.get("token") ?? "").trim().toUpperCase();
   187→   187→    if (q.length === 6 && /^[A-Z0-9]{6}$/.test(q) && autoTriedRef.current !== q) {
   188→   188→      autoTriedRef.current = q;
   189→   189→      fillToken(q);
   190→   190→      activeTokenRef.current = q;
   191→   191→      void validateToken(q);
   192→   192→    }
   193→   193→  }, [searchParams, fillToken, validateToken]);
   194→   194→
   195→   195→  /** Hitung mundur TTL token — hanya vote PERTAMA (re-vote tanpa TTL) */
   196→   196→  useEffect(() => {
   197→   197→    if (!session || !session.expiresAt || (phase !== "ballot" && phase !== "review")) return;
   198→   198→    const tick = () => {
   199→   199→      const left = new Date(session.expiresAt!).getTime() - Date.now();
   200→   200→      setRemainingMs(Math.max(0, left));
   201→   201→      if (left <= 0) {
   202→   202→        setPhase("gate");
   203→   203→        setSession(null);
   204→   204→        setBallots([]);
   205→   205→        setSelections({});
   206→   206→        setTokenChars(Array(6).fill(""));
   207→   207→        setGateError("Token kedaluwarsa saat berada di bilik. Minta token baru di pos bantuan PPUR.");
   208→   208→      }
   209→   209→    };
   210→   210→    tick();
   211→   211→    const t = setInterval(tick, 1000);
   212→   212→    return () => clearInterval(t);
   213→   213→  }, [session, phase]);
   214→   214→
   215→   215→  const resetAll = useCallback(() => {
   216→   216→    setPhase("gate");
   217→   217→    setTokenChars(Array(6).fill(""));
   218→   218→    setGateError(null);
   219→   219→    setSession(null);
   220→   220→    setBallots([]);
   221→   221→    setStep(0);
   222→   222→    setSelections({});
   223→   223→    setSubmitError(null);
   224→   224→    setReceiptCode(null);
   225→   225→    setIsRevote(false);
   226→   226→    activeTokenRef.current = "";
   227→   227→  }, []);
   228→   228→
   229→   229→  const submitVote = useCallback(async () => {
   230→   230→    if (!session) return;
   231→   231→    setPhase("submitting");
   232→   232→    setSubmittingMsg(isRevote ? "Mengganti Suara Anda..." : "Mengenkripsi & Mengacak Suara...");
   233→   233→    try {
   234→   234→      const res = await fetch("/api/pemilih/submit", {
   235→   235→        method: "POST",
   236→   236→        headers: { "Content-Type": "application/json" },
   237→   237→        body: JSON.stringify({
   238→   238→          token: activeTokenRef.current,
   239→   239→          selections: {
   240→   240→            presmaCandidateId: selections["PRESMA"],
   241→   241→            dpmCandidateId: selections["DPM"],
   242→   242→            himaCandidateId: selections["HIMA"] === "EMPTY" ? null : selections["HIMA"],
   243→   243→          },
   244→   244→        }),
   245→   245→      });
   246→   246→      const json = await res.json();
   247→   247→      if (!json.ok) {
   248→   248→        setSubmitError(json.message ?? "Gagal mengirim suara.");
   249→   249→        setPhase("review");
   250→   250→        return;
   251→   251→      }
   252→   252→      setReceiptCode(json.receiptCode ?? null);
   253→   253→      setVoteVersion(json.voteVersion ?? 1);
   254→   254→      setIsRevote(Boolean(json.isRevote));
   255→   255→      setPhase("success");
   256→   256→    } catch {
   257→   257→      setSubmitError("Koneksi gagal saat mengirim suara. Suara tidak terkirim — coba lagi atau hubungi pos bantuan.");
   258→   258→      setPhase("review");
   259→   259→    }
   260→   260→  }, [session, selections, isRevote]);
   261→   261→
   262→   262→  const lockVote = useCallback(async () => {
   263→   263→    setLockBusy(true);
   264→   264→    try {
   265→   265→      const res = await fetch("/api/pemilih/lock", {
   266→   266→        method: "POST",
   267→   267→        headers: { "Content-Type": "application/json" },
   268→   268→        body: JSON.stringify({ token: activeTokenRef.current }),
   269→   269→      });
   270→   270→      const json = await res.json();
   271→   271→      if (json.ok) {
   272→   272→        setPhase("locked");
   273→   273→      } else {
   274→   274→        setGateError(json.message ?? "Gagal mengunci suara.");
   275→   275→      }
   276→   276→    } catch {
   277→   277→      setGateError("Koneksi gagal saat mengunci suara.");
   278→   278→    } finally {
   279→   279→      setLockBusy(false);
   280→   280→    }
   281→   281→  }, []);
   282→   282→
   283→   283→  const revoteAgain = useCallback(async () => {
   284→   284→    // Buka bilik ulang dengan token yang sama — suara baru akan MENGGANTI yang lama
   285→   285→    await validateToken(activeTokenRef.current);
   286→   286→  }, [validateToken]);
   287→   287→
   288→   288→  /** Keyboard fisik juga didukung (desktop) */
   289→   289→  const pressKey = useCallback((key: string) => {
   290→   290→    setTokenChars((prev) => {
   291→   291→      const idx = prev.findIndex((c) => c === "");
   292→   292→      if (idx === -1) return prev;
   293→   293→      const next = [...prev];
   294→   294→      next[idx] = key;
   295→   295→      return next;
   296→   296→    });
   297→   297→    setGateError(null);
   298→   298→  }, []);
   299→   299→
   300→   300→  const backspace = useCallback(() => {
   301→   301→    setTokenChars((prev) => {
   302→   302→      const next = [...prev];
   303→   303→      const idx = next.findIndex((c) => c === "");
   304→   304→      const target = idx === -1 ? 5 : idx - 1;
   305→   305→      if (target >= 0) next[target] = "";
   306→   306→      return next;
   307→   307→    });
   308→   308→  }, []);
   309→   309→
   310→   310→  useEffect(() => {
   311→   311→    if (phase !== "gate") return;
   312→   312→    const handler = (e: KeyboardEvent) => {
   313→   313→      if (/^[a-zA-Z0-9]$/.test(e.key)) pressKey(e.key.toUpperCase());
   314→   314→      else if (e.key === "Backspace") backspace();
   315→   315→    };
   316→   316→    window.addEventListener("keydown", handler);
   317→   317→    return () => window.removeEventListener("keydown", handler);
   318→   318→  }, [phase, pressKey, backspace]);
   319→   319→
   320→   320→  const fullToken = tokenChars.join("");
   321→   321→  const minutes = Math.floor(remainingMs / 60000);
   322→   322→  const seconds = Math.floor((remainingMs % 60000) / 1000);
   323→   323→  const timeCritical = remainingMs > 0 && remainingMs < 120_000;
   324→   324→  const hasTtl = Boolean(session?.expiresAt);
   325→   325→
   326→   326→  const keypadKeys = useMemo(() => {
   327→   327→    const digits = ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"];
   328→   328→    const letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
   329→   329→    return [...digits, ...letters];
   330→   330→  }, []);
   331→   331→
   332→   332→  // ============================ GATE ============================
   333→   333→  if (phase === "gate") {
   334→   334→    return (
   335→   335→      <div className="flex min-h-screen flex-col items-center justify-center px-4 py-10">
   336→   336→        <div className="w-full max-w-md">
   337→   337→          <div className="mb-7 text-center">
   338→   338→            <div className="mx-auto mb-5 flex h-16 w-16 items-center justify-center rounded-[14px] border-2 border-[#141E19] bg-[#1B7F4E] shadow-[4px_4px_0px_#141E19]">
   339→   339→              <Lock className="h-7 w-7 text-white" />
   340→   340→            </div>
   341→   341→            <h1 className="font-display text-3xl font-extrabold tracking-[-0.03em] md:text-4xl">
   342→   342→              PORTAL PEMILIH <span className="text-[#1B7F4E]">PEMIRA ONLINE</span>
   343→   343→            </h1>
   344→   344→            <p className="mt-2 text-sm font-medium text-[#4A5851]">
   345→   345→              Pemira Poltek CWE 100% Online — pilih dari perangkatmu sendiri
   346→   346→            </p>
   347→   347→          </div>
   348→   348→
   349→   349→          <div className="mb-4 flex items-center gap-2 rounded-[10px] border border-[#D8D3C8] bg-white px-4 py-3">
   350→   350→            <ScanLine className="h-5 w-5 shrink-0 text-[#1B7F4E]" />
   351→   351→            <p className="text-[12.5px] leading-relaxed text-[#4A5851]">
   352→   352→              <strong className="text-[#141E19]">Cara masuk:</strong> scan <strong>QR code</strong> pada slip token dari pos

   354→   354→            </p>
   355→   355→          </div>
   356→   356→
   357→   357→          {/* 6 Kotak Token */}
   358→   358→          <div className="mb-5 flex justify-center gap-2" role="group" aria-label="Kotak input token">
   359→   359→            {tokenChars.map((c, i) => (
   360→   360→              <div
   361→   361→                key={i}
   362→   362→                className={`flex h-[68px] w-[52px] items-center justify-center rounded-[10px] border-2 bg-white font-mono text-2xl font-bold transition-shadow sm:h-[76px] sm:w-[58px] sm:text-3xl ${
   363→   363→                  c ? "border-[#141E19] shadow-[3px_3px_0px_#141E19]" : "border-[#141E19]/40"
   364→   364→                }`}
   365→   365→                aria-label={`Karakter ${i + 1}: ${c || "kosong"}`}
   366→   366→              >
   367→   367→                {c}
   368→   368→              </div>
   369→   369→            ))}
   370→   370→          </div>
   371→   371→
   372→   372→          {gateError && (
   373→   373→            <div className="mx-auto mb-4 flex items-start gap-2 rounded-[10px] border-2 border-[#E05A47] bg-[#FDE8E5] px-4 py-3" role="alert">
   374→   374→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-[#842029]" />
   375→   375→              <p className="text-sm font-semibold text-[#842029]">{gateError}</p>
   376→   376→            </div>
   377→   377→          )}
   378→   378→
   379→   379→          {/* Keypad virtual (mobile-first) */}
   380→   380→          <div className="tactile-card mx-auto p-4">
   381→   381→            <div className="grid grid-cols-6 gap-1.5">
   382→   382→              {keypadKeys.map((k) => (
   383→   383→                <button
   384→   384→                  key={k}
   385→   385→                  type="button"
   386→   386→                  onClick={() => pressKey(k)}
   387→   387→                  className="h-10 rounded-lg border-2 border-[#141E19] bg-white font-mono text-sm font-bold shadow-[2px_2px_0px_#141E19] transition-all hover:-translate-y-0.5 hover:shadow-[3px_3px_0px_#141E19] active:translate-y-0.5 active:shadow-none sm:h-11"
   388→   388→                >
   389→   389→                  {k}
   390→   390→                </button>
   391→   391→              ))}
   392→   392→              <button
   393→   393→                type="button"
   394→   394→                onClick={backspace}
   395→   395→                className="col-span-2 flex h-10 items-center justify-center gap-1.5 rounded-lg border-2 border-[#141E19] bg-[#F2B705] text-sm font-bold shadow-[2px_2px_0px_#141E19] transition-all hover:-translate-y-0.5 active:translate-y-0.5 active:shadow-none sm:h-11"
   396→   396→                aria-label="Hapus karakter terakhir"
   397→   397→              >

   401→   401→
   402→   402→            <button
   403→   403→              type="button"

   405→   405→              onClick={() => {
