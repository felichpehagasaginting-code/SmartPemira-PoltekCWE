   193→   193→export function PemilihPortal() {
   194→   194→  const searchParams = useSearchParams();
   195→   195→  const [phase, setPhase] = useState<Phase>("gate");
   196→   196→  const [tokenChars, setTokenChars] = useState<string[]>(Array(6).fill(""));
   197→   197→  const [gateError, setGateError] = useState<string | null>(null);
   198→   198→  const [validating, setValidating] = useState(false);
   199→   199→  const [session, setSession] = useState<KioskSession | null>(null);
   200→   200→  const [isRevote, setIsRevote] = useState(false);
   201→   201→  const [ballots, setBallots] = useState<BallotPayload[]>([]);
   202→   202→  const [step, setStep] = useState(0);
   203→   203→  const [selections, setSelections] = useState<Selections>({});
   204→   204→  const [submitError, setSubmitError] = useState<string | null>(null);
   205→   205→  const [remainingMs, setRemainingMs] = useState(0);
   206→   206→  const [submittingMsg, setSubmittingMsg] = useState("Mengenkripsi & Mengacak Suara...");
   207→   207→  const [receiptCode, setReceiptCode] = useState<string | null>(null);
   208→   208→  const [voteVersion, setVoteVersion] = useState(1);
   209→   209→  const [lockBusy, setLockBusy] = useState(false);
   210→   210→  const activeTokenRef = useRef<string>("");
   211→   211→  const autoTriedRef = useRef<string>("");
   212→   212→
   213→   213→  const current = ballots[step];
   214→   214→
   215→   215→  const fillToken = useCallback((token: string) => {
   216→   216→    setTokenChars(token.slice(0, 6).padEnd(6, " ").split("").map((c) => (c === " " ? "" : c)));
   217→   217→  }, []);
   218→   218→
   219→   219→  const validateToken = useCallback(async (token: string) => {
   220→   220→    setValidating(true);
   221→   221→    setGateError(null);
   222→   222→    try {
   223→   223→      const res = await fetch("/api/pemilih/validate", {
   224→   224→        method: "POST",
   225→   225→        headers: { "Content-Type": "application/json" },
   226→   226→        body: JSON.stringify({ token }),
   227→   227→      });
   228→   228→      const json = await res.json();
   229→   229→      if (!json.ok) {
   230→   230→        setGateError(json.message ?? "Token tidak sah.");
   231→   231→        return false;
   232→   232→      }
   233→   233→      setSession(json.session);
   234→   234→      setBallots(json.ballots);
   235→   235→      setIsRevote(Boolean(json.isRevote));
   236→   236→      setVoteVersion(json.voteVersion ?? 1);
   237→   237→      setSelections({});
   238→   238→      setStep(0);
   239→   239→      setPhase("ballot");
   240→   240→      return true;
   241→   241→    } catch {
   242→   242→      setGateError("Koneksi ke server bilik suara gagal.");
   243→   243→      return false;
   244→   244→    } finally {
   245→   245→      setValidating(false);
   246→   246→    }
   247→   247→  }, []);
   248→   248→
   249→   249→  // Auto-isi & auto-validasi token dari QR (?token=XXXXXX pada slip PPUR)
   250→   250→  useEffect(() => {
   251→   251→    const q = (searchParams.get("token") ?? "").trim().toUpperCase();
   252→   252→    if (q.length === 6 && /^[A-Z0-9]{6}$/.test(q) && autoTriedRef.current !== q) {
   253→   253→      autoTriedRef.current = q;
   254→   254→      fillToken(q);
   255→   255→      activeTokenRef.current = q;
   256→   256→      void validateToken(q);
   257→   257→    }
   258→   258→  }, [searchParams, fillToken, validateToken]);
   259→   259→
   260→   260→  /** Hitung mundur TTL token — hanya vote PERTAMA (re-vote tanpa TTL) */
   261→   261→  useEffect(() => {
   262→   262→    if (!session || !session.expiresAt || (phase !== "ballot" && phase !== "review")) return;
   263→   263→    const tick = () => {
   264→   264→      const left = new Date(session.expiresAt!).getTime() - Date.now();
   265→   265→      setRemainingMs(Math.max(0, left));
   266→   266→      if (left <= 0) {
   267→   267→        setPhase("gate");
   268→   268→        setSession(null);
   269→   269→        setBallots([]);
   270→   270→        setSelections({});
   271→   271→        setTokenChars(Array(6).fill(""));
   272→   272→        setGateError("Token kedaluwarsa saat berada di bilik. Minta token baru di pos bantuan PPUR.");
   273→   273→      }
   274→   274→    };
   275→   275→    tick();
   276→   276→    const t = setInterval(tick, 1000);
   277→   277→    return () => clearInterval(t);
   278→   278→  }, [session, phase]);
   279→   279→
   280→   280→  const resetAll = useCallback(() => {
   281→   281→    setPhase("gate");
   282→   282→    setTokenChars(Array(6).fill(""));
   283→   283→    setGateError(null);
   284→   284→    setSession(null);
   285→   285→    setBallots([]);
   286→   286→    setStep(0);
   287→   287→    setSelections({});
   288→   288→    setSubmitError(null);
   289→   289→    setReceiptCode(null);
   290→   290→    setIsRevote(false);
   291→   291→    activeTokenRef.current = "";
   292→   292→  }, []);
   293→   293→
   294→   294→  const submitVote = useCallback(async () => {
   295→   295→    if (!session) return;
   296→   296→    setPhase("submitting");
   297→   297→    setSubmittingMsg(isRevote ? "Mengganti Suara Anda..." : "Mengenkripsi & Mengacak Suara...");
   298→   298→    try {
   299→   299→      const res = await fetch("/api/pemilih/submit", {
   300→   300→        method: "POST",
   301→   301→        headers: { "Content-Type": "application/json" },
   302→   302→        body: JSON.stringify({
   303→   303→          token: activeTokenRef.current,
   304→   304→          selections: {
   305→   305→            presmaCandidateId: selections["PRESMA"],
   306→   306→            dpmCandidateId: selections["DPM"],
   307→   307→            himaCandidateId: selections["HIMA"] === "EMPTY" ? null : selections["HIMA"],
   308→   308→          },
   309→   309→        }),
   310→   310→      });
   311→   311→      const json = await res.json();
   312→   312→      if (!json.ok) {
   313→   313→        setSubmitError(json.message ?? "Gagal mengirim suara.");
   314→   314→        setPhase("review");
   315→   315→        return;
   316→   316→      }
   317→   317→      setReceiptCode(json.receiptCode ?? null);
   318→   318→      setVoteVersion(json.voteVersion ?? 1);
   319→   319→      setIsRevote(Boolean(json.isRevote));
   320→   320→      setPhase("success");
   321→   321→    } catch {
   322→   322→      setSubmitError("Koneksi gagal saat mengirim suara. Suara tidak terkirim — coba lagi atau hubungi pos bantuan.");
   323→   323→      setPhase("review");
   324→   324→    }
   325→   325→  }, [session, selections, isRevote]);
   326→   326→
   327→   327→  const lockVote = useCallback(async () => {
   328→   328→    setLockBusy(true);
   329→   329→    try {
   330→   330→      const res = await fetch("/api/pemilih/lock", {
   331→   331→        method: "POST",
   332→   332→        headers: { "Content-Type": "application/json" },
   333→   333→        body: JSON.stringify({ token: activeTokenRef.current }),
   334→   334→      });
   335→   335→      const json = await res.json();
   336→   336→      if (json.ok) {
   337→   337→        setPhase("locked");
   338→   338→      } else {
   339→   339→        setGateError(json.message ?? "Gagal mengunci suara.");
   340→   340→      }
   341→   341→    } catch {
   342→   342→      setGateError("Koneksi gagal saat mengunci suara.");
   343→   343→    } finally {
   344→   344→      setLockBusy(false);
   345→   345→    }
   346→   346→  }, []);
   347→   347→
   348→   348→  const revoteAgain = useCallback(async () => {
   349→   349→    // Buka bilik ulang dengan token yang sama — suara baru akan MENGGANTI yang lama
   350→   350→    await validateToken(activeTokenRef.current);
   351→   351→  }, [validateToken]);
   352→   352→
   353→   353→  /** Keyboard fisik juga didukung (desktop) */
   354→   354→  const pressKey = useCallback((key: string) => {
   355→   355→    setTokenChars((prev) => {
   356→   356→      const idx = prev.findIndex((c) => c === "");
   357→   357→      if (idx === -1) return prev;
   358→   358→      const next = [...prev];
   359→   359→      next[idx] = key;
   360→   360→      return next;
   361→   361→    });
   362→   362→    setGateError(null);
   363→   363→  }, []);
   364→   364→
   365→   365→  const backspace = useCallback(() => {
   366→   366→    setTokenChars((prev) => {
   367→   367→      const next = [...prev];
   368→   368→      const idx = next.findIndex((c) => c === "");
   369→   369→      const target = idx === -1 ? 5 : idx - 1;
   370→   370→      if (target >= 0) next[target] = "";
   371→   371→      return next;
   372→   372→    });
   373→   373→  }, []);
   374→   374→
   375→   375→  useEffect(() => {
   376→   376→    if (phase !== "gate") return;
   377→   377→    const handler = (e: KeyboardEvent) => {
   378→   378→      if (/^[a-zA-Z0-9]$/.test(e.key)) pressKey(e.key.toUpperCase());
   379→   379→      else if (e.key === "Backspace") backspace();
   380→   380→    };
   381→   381→    window.addEventListener("keydown", handler);
   382→   382→    return () => window.removeEventListener("keydown", handler);
   383→   383→  }, [phase, pressKey, backspace]);
   384→   384→
   385→   385→  const fullToken = tokenChars.join("");
   386→   386→  const minutes = Math.floor(remainingMs / 60000);
   387→   387→  const seconds = Math.floor((remainingMs % 60000) / 1000);
   388→   388→  const timeCritical = remainingMs > 0 && remainingMs < 120_000;
   389→   389→  const hasTtl = Boolean(session?.expiresAt);
   390→   390→
   391→   391→  const keypadKeys = useMemo(() => {
   392→   392→    const digits = ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"];
   393→   393→    const letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
   394→   394→    return [...digits, ...letters];
   395→   395→  }, []);
   396→   396→
   397→   397→  // ============================ GATE ============================
   398→   398→  if (phase === "gate") {
   399→   399→    return (
   400→   400→      <div className="flex min-h-screen flex-col items-center justify-center px-4 py-10">
   401→   401→        <div className="w-full max-w-md">
   402→   402→          <div className="mb-7 text-center">
   403→   403→            <div className="mx-auto mb-5 flex h-16 w-16 items-center justify-center rounded-[14px] border-2 border-[#141E19] bg-[#1B7F4E] shadow-[4px_4px_0px_#141E19]">
   404→   404→              <Lock className="h-7 w-7 text-white" />
   405→   405→            </div>
   406→   406→            <h1 className="font-display text-3xl font-extrabold tracking-[-0.03em] md:text-4xl">
   407→   407→              PORTAL PEMILIH <span className="text-[#1B7F4E]">PEMIRA ONLINE</span>
   408→   408→            </h1>
   409→   409→            <p className="mt-2 text-sm font-medium text-[#4A5851]">
   410→   410→              Pemira Poltek CWE 100% Online — pilih dari perangkatmu sendiri
   411→   411→            </p>
   412→   412→          </div>
   413→   413→
   414→   414→          <div className="mb-4 flex items-center gap-2 rounded-[10px] border border-[#D8D3C8] bg-white px-4 py-3">
   415→   415→            <ScanLine className="h-5 w-5 shrink-0 text-[#1B7F4E]" />
