    34→/** QR asli (Rekomendasi #9): encode URL verifikasi publik dokumen */
    35→function RealQr({ text, size = 132 }: { text: string; size?: number }) {
    36→  const [dataUrl, setDataUrl] = useState<string | null>(null);
    37→  useEffect(() => {
    38→    let alive = true;
    39→    QRCode.toDataURL(text, { width: size * 2, margin: 1, color: { dark: "#141E19", light: "#FFFFFF" } })
    40→      .then((url) => alive && setDataUrl(url))
    41→      .catch(() => alive && setDataUrl(null));
    42→    return () => {
    43→      alive = false;
    44→    };
    45→  }, [text, size]);
    46→  if (!dataUrl) {
    47→    return <Loader2 className="h-8 w-8 animate-spin text-[#7A8881]" />;
    48→  }
    49→  return <img src={dataUrl} width={size} height={size} alt="QR verifikasi Berita Acara" className="rounded-[4px]" />;
    50→}
