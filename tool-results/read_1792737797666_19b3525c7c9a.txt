    44→  const initials = name === "KOTAK KOSONG" ? "□" : name.split(" ").slice(0, 2).map((w) => w[0]).join("").toUpperCase();
    45→  if (photo) {

    48→        { }
    49→        <img src={photo} alt={`Foto ${name}`} className="h-full w-full object-cover" />
    50→      </div>

    54→    <div

    64→/** Label peran resmi per kategori — DPM punya struktur 3 peran (Ketua Umum, Ketua 1, Ketua 2) */
    65→function memberRoles(category: string): { leader: string; vice: string; second?: string } {
    66→  if (category === "DPM") return { leader: "Ketua Umum", vice: "Ketua 1", second: "Ketua 2" };
    67→  if (category === "PRESMA") return { leader: "Presma", vice: "Wakil Presma" };
    68→  return { leader: "Ketua", vice: "Wakil Ketua" };
    69→}

   193→export function PemilihPortal() {
   194→  const searchParams = useSearchParams();
   195→  const [phase, setPhase] = useState<Phase>("gate");
   196→  const [tokenChars, setTokenChars] = useState<string[]>(Array(6).fill(""));
   197→  const [gateError, setGateError] = useState<string | null>(null);
   198→  const [validating, setValidating] = useState(false);
   199→  const [session, setSession] = useState<KioskSession | null>(null);
   200→  const [isRevote, setIsRevote] = useState(false);
   201→  const [ballots, setBallots] = useState<BallotPayload[]>([]);
   202→  const [step, setStep] = useState(0);
   203→  const [selections, setSelections] = useState<Selections>({});
   204→  const [submitError, setSubmitError] = useState<string | null>(null);
   205→  const [remainingMs, setRemainingMs] = useState(0);
   206→  const [submittingMsg, setSubmittingMsg] = useState("Mengenkripsi & Mengacak Suara...");
   207→  const [receiptCode, setReceiptCode] = useState<string | null>(null);
   208→  const [voteVersion, setVoteVersion] = useState(1);
   209→  const [lockBusy, setLockBusy] = useState(false);
   210→  const activeTokenRef = useRef<string>("");
   211→  const autoTriedRef = useRef<string>("");
   212→
   213→  const current = ballots[step];
   214→
   215→  const fillToken = useCallback((token: string) => {
   216→    setTokenChars(token.slice(0, 6).padEnd(6, " ").split("").map((c) => (c === " " ? "" : c)));
   217→  }, []);
   218→
   219→  const validateToken = useCallback(async (token: string) => {
   220→    setValidating(true);
   221→    setGateError(null);
   222→    try {
   223→      const res = await fetch("/api/pemilih/validate", {
   224→        method: "POST",
   225→        headers: { "Content-Type": "application/json" },
   226→        body: JSON.stringify({ token }),
   227→      });
   228→      const json = await res.json();
   229→      if (!json.ok) {
   230→        setGateError(json.message ?? "Token tidak sah.");
   231→        return false;
   232→      }
   233→      setSession(json.session);
   234→      setBallots(json.ballots);
   235→      setIsRevote(Boolean(json.isRevote));
   236→      setVoteVersion(json.voteVersion ?? 1);
   237→      setSelections({});
   238→      setStep(0);
   239→      setPhase("ballot");
   240→      return true;
   241→    } catch {
   242→      setGateError("Koneksi ke server bilik suara gagal.");
   243→      return false;
   244→    } finally {
   245→      setValidating(false);
   246→    }
   247→  }, []);
   248→
   249→  // Auto-isi & auto-validasi token dari QR (?token=XXXXXX pada slip PPUR)
   250→  useEffect(() => {
   251→    const q = (searchParams.get("token") ?? "").trim().toUpperCase();
   252→    if (q.length === 6 && /^[A-Z0-9]{6}$/.test(q) && autoTriedRef.current !== q) {
   253→      autoTriedRef.current = q;
   254→      fillToken(q);
   255→      activeTokenRef.current = q;
   256→      void validateToken(q);
   257→    }
   258→  }, [searchParams, fillToken, validateToken]);
   259→
   260→  /** Hitung mundur TTL token — hanya vote PERTAMA (re-vote tanpa TTL) */
   261→  useEffect(() => {
   262→    if (!session || !session.expiresAt || (phase !== "ballot" && phase !== "review")) return;
   263→    const tick = () => {
   264→      const left = new Date(session.expiresAt!).getTime() - Date.now();
   265→      setRemainingMs(Math.max(0, left));
   266→      if (left <= 0) {
   267→        setPhase("gate");
   268→        setSession(null);
   269→        setBallots([]);
   270→        setSelections({});
   271→        setTokenChars(Array(6).fill(""));
   272→        setGateError("Token kedaluwarsa saat berada di bilik. Minta token baru di pos bantuan PPUR.");
   273→      }
   274→    };
   275→    tick();
   276→    const t = setInterval(tick, 1000);
   277→    return () => clearInterval(t);
   278→  }, [session, phase]);
   279→
   280→  const resetAll = useCallback(() => {
   281→    setPhase("gate");
   282→    setTokenChars(Array(6).fill(""));
   283→    setGateError(null);
   284→    setSession(null);
   285→    setBallots([]);
   286→    setStep(0);
   287→    setSelections({});
   288→    setSubmitError(null);
   289→    setReceiptCode(null);
   290→    setIsRevote(false);
   291→    activeTokenRef.current = "";
   292→  }, []);
   293→
   294→  const submitVote = useCallback(async () => {
   295→    if (!session) return;
   296→    setPhase("submitting");
   297→    setSubmittingMsg(isRevote ? "Mengganti Suara Anda..." : "Mengenkripsi & Mengacak Suara...");
   298→    try {
   299→      const res = await fetch("/api/pemilih/submit", {
   300→        method: "POST",
   301→        headers: { "Content-Type": "application/json" },
   302→        body: JSON.stringify({
   303→          token: activeTokenRef.current,
   304→          selections: {
   305→            presmaCandidateId: selections["PRESMA"],
   306→            dpmCandidateId: selections["DPM"],
   307→            himaCandidateId: selections["HIMA"] === "EMPTY" ? null : selections["HIMA"],
   308→          },
   309→        }),
   310→      });
   311→      const json = await res.json();
   312→      if (!json.ok) {
   313→        setSubmitError(json.message ?? "Gagal mengirim suara.");
   314→        setPhase("review");
   315→        return;
   316→      }
   317→      setReceiptCode(json.receiptCode ?? null);
   318→      setVoteVersion(json.voteVersion ?? 1);
   319→      setIsRevote(Boolean(json.isRevote));
   320→      setPhase("success");
   321→    } catch {
   322→      setSubmitError("Koneksi gagal saat mengirim suara. Suara tidak terkirim — coba lagi atau hubungi pos bantuan.");
   323→      setPhase("review");
   324→    }
   325→  }, [session, selections, isRevote]);
   326→
   327→  const lockVote = useCallback(async () => {
   328→    setLockBusy(true);
   329→    try {
   330→      const res = await fetch("/api/pemilih/lock", {
   331→        method: "POST",
   332→        headers: { "Content-Type": "application/json" },
   333→        body: JSON.stringify({ token: activeTokenRef.current }),
   334→      });
   335→      const json = await res.json();
   336→      if (json.ok) {
   337→        setPhase("locked");
   338→      } else {
   339→        setGateError(json.message ?? "Gagal mengunci suara.");
   340→      }
   341→    } catch {
   342→      setGateError("Koneksi gagal saat mengunci suara.");
   343→    } finally {
   344→      setLockBusy(false);
   345→    }
   346→  }, []);
   347→
   348→  const revoteAgain = useCallback(async () => {
   349→    // Buka bilik ulang dengan token yang sama — suara baru akan MENGGANTI yang lama
   350→    await validateToken(activeTokenRef.current);
   351→  }, [validateToken]);
   352→
   353→  /** Keyboard fisik juga didukung (desktop) */
   354→  const pressKey = useCallback((key: string) => {
   355→    setTokenChars((prev) => {
   356→      const idx = prev.findIndex((c) => c === "");
   357→      if (idx === -1) return prev;
   358→      const next = [...prev];
   359→      next[idx] = key;
   360→      return next;
   361→    });
   362→    setGateError(null);
   363→  }, []);
   364→
   365→  const backspace = useCallback(() => {
   366→    setTokenChars((prev) => {
   367→      const next = [...prev];
   368→      const idx = next.findIndex((c) => c === "");
   369→      const target = idx === -1 ? 5 : idx - 1;
   370→      if (target >= 0) next[target] = "";
   371→      return next;
   372→    });
   373→  }, []);
   374→
   375→  useEffect(() => {
   376→    if (phase !== "gate") return;
   377→    const handler = (e: KeyboardEvent) => {
   378→      if (/^[a-zA-Z0-9]$/.test(e.key)) pressKey(e.key.toUpperCase());
   379→      else if (e.key === "Backspace") backspace();
   380→    };
   381→    window.addEventListener("keydown", handler);
   382→    return () => window.removeEventListener("keydown", handler);
   383→  }, [phase, pressKey, backspace]);
   384→
   385→  const fullToken = tokenChars.join("");
   386→  const minutes = Math.floor(remainingMs / 60000);
   387→  const seconds = Math.floor((remainingMs % 60000) / 1000);
   388→  const timeCritical = remainingMs > 0 && remainingMs < 120_000;
   389→  const hasTtl = Boolean(session?.expiresAt);
   390→
   391→  const keypadKeys = useMemo(() => {
   392→    const digits = ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"];
   393→    const letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
   394→    return [...digits, ...letters];
   395→  }, []);
   396→
   397→  // ============================ GATE ============================
   398→  if (phase === "gate") {
   399→    return (
   400→      <div className="flex min-h-screen flex-col items-center justify-center px-4 py-10">
   401→        <div className="w-full max-w-md">
   402→          <div className="mb-7 text-center">
   403→            <div className="mx-auto mb-5 flex h-16 w-16 items-center justify-center rounded-[14px] border-2 border-[#141E19] bg-[#1B7F4E] shadow-[4px_4px_0px_#141E19]">
   404→              <Lock className="h-7 w-7 text-white" />
   405→            </div>
   406→            <h1 className="font-display text-3xl font-extrabold tracking-[-0.03em] md:text-4xl">
   407→              PORTAL PEMILIH <span className="text-[#1B7F4E]">PEMIRA ONLINE</span>
   408→            </h1>
   409→            <p className="mt-2 text-sm font-medium text-[#4A5851]">
   410→              Pemira Poltek CWE 100% Online — pilih dari perangkatmu sendiri
   411→            </p>
   412→          </div>
   413→
   414→          <div className="mb-4 flex items-center gap-2 rounded-[10px] border border-[#D8D3C8] bg-white px-4 py-3">
   415→            <ScanLine className="h-5 w-5 shrink-0 text-[#1B7F4E]" />
   416→            <p className="text-[12.5px] leading-relaxed text-[#4A5851]">
   417→              <strong className="text-[#141E19]">Cara masuk:</strong> scan <strong>QR code</strong> pada slip token dari pos
   418→              bantuan PPUR, atau ketik 6 karakter token di bawah.
   419→            </p>
   420→          </div>
   421→
   422→          {/* 6 Kotak Token */}
   423→          <div className="mb-5 flex justify-center gap-2" role="group" aria-label="Kotak input token">
   424→            {tokenChars.map((c, i) => (
   425→              <div
   426→                key={i}
   427→                className={`flex h-[68px] w-[52px] items-center justify-center rounded-[10px] border-2 bg-white font-mono text-2xl font-bold transition-shadow sm:h-[76px] sm:w-[58px] sm:text-3xl ${
   428→                  c ? "border-[#141E19] shadow-[3px_3px_0px_#141E19]" : "border-[#141E19]/40"
   429→                }`}
   430→                aria-label={`Karakter ${i + 1}: ${c || "kosong"}`}
   431→              >
   432→                {c}
   433→              </div>
   434→            ))}
   435→          </div>
   436→
   437→          {gateError && (
   438→            <div className="mx-auto mb-4 flex items-start gap-2 rounded-[10px] border-2 border-[#E05A47] bg-[#FDE8E5] px-4 py-3" role="alert">
   439→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-[#842029]" />
   440→              <p className="text-sm font-semibold text-[#842029]">{gateError}</p>
   441→            </div>
   442→          )}
   443→
   444→          {/* Keypad virtual (mobile-first) */}
   445→          <div className="tactile-card mx-auto p-4">
   446→            <div className="grid grid-cols-6 gap-1.5">
   447→              {keypadKeys.map((k) => (
   448→                <button
   449→                  key={k}
   450→                  type="button"
   451→                  onClick={() => pressKey(k)}
   452→                  className="h-10 rounded-lg border-2 border-[#141E19] bg-white font-mono text-sm font-bold shadow-[2px_2px_0px_#141E19] transition-all hover:-translate-y-0.5 hover:shadow-[3px_3px_0px_#141E19] active:translate-y-0.5 active:shadow-none sm:h-11"
   453→                >
   454→                  {k}
   455→                </button>
   456→              ))}
   457→              <button
   458→                type="button"
   459→                onClick={backspace}
   460→                className="col-span-2 flex h-10 items-center justify-center gap-1.5 rounded-lg border-2 border-[#141E19] bg-[#F2B705] text-sm font-bold shadow-[2px_2px_0px_#141E19] transition-all hover:-translate-y-0.5 active:translate-y-0.5 active:shadow-none sm:h-11"
   461→                aria-label="Hapus karakter terakhir"
   462→              >
   463→                <Delete className="h-4 w-4" /> HAPUS
   464→              </button>
   465→            </div>
   466→
   467→            <button
   468→              type="button"
   469→              disabled={fullToken.length !== 6 || validating}
   470→              onClick={() => {
   471→                activeTokenRef.current = fullToken;
   472→                validateToken(fullToken);
   473→              }}
   474→              className="tactile-btn mt-4 flex h-14 w-full items-center justify-center gap-2 text-base font-extrabold tracking-wide"
   475→            >
   476→              {validating ? "MEMVERIFIKASI TOKEN..." : "MASUK BILIK SUARA"}
   477→              {!validating && <ArrowRight className="h-5 w-5" />}
   478→            </button>
   479→          </div>
   480→
